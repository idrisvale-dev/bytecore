local function formatPercent(score)
    return string.format("%.2f%%", math.clamp(score, 0, 100))
end

local function formatTime(seconds)
    if seconds < 0.001 then
        return string.format("%.2fμs", seconds * 1000000)
    elseif seconds < 1 then
        return string.format("%.2fms", seconds * 1000)
    else
        return string.format("%.2fs", seconds)
    end
end

local function repeatChar(char, count)
    local s = ""
    for i = 1, count do
        s = s .. char
    end
    return s
end

local function centerText(text, width)
    local padding = math.max(0, width - #text)
    local left = math.floor(padding / 2)
    local right = padding - left
    return repeatChar(" ", left) .. text .. repeatChar(" ", right)
end

local function getStatusSymbol(status)
    if status == ResultStatus.PASS then
        return BenchmarkConfig.PassSymbol
    elseif status == ResultStatus.FAIL then
        return BenchmarkConfig.FailSymbol
    elseif status == ResultStatus.UNSUPPORTED then
        return BenchmarkConfig.UnsupportedSymbol
    elseif status == ResultStatus.ERROR then
        return BenchmarkConfig.ErrorSymbol
    elseif status == ResultStatus.TIMEOUT then
        return BenchmarkConfig.TimeoutSymbol
    end
    return "?"
end

local function safeTypeof(value)
    local ok, result = pcall(function()
        return typeof(value)
    end)
    if ok then return result end
    ok, result = pcall(function()
        return type(value)
    end)
    if ok then return result end
    return "unknown"
end

local SafeExec = {}

function SafeExec.run(fn, timeout)
    timeout = timeout or BenchmarkConfig.TimeoutPerTest
    local startTime = os.clock()
    local co = coroutine.create(function()
        return fn()
    end)
    
    local deadline = startTime + timeout
    local results = nil
    local err = nil
    local timedOut = false
    
    local ok, ret = coroutine.resume(co)
    
    if not ok then
        err = tostring(ret)
        local elapsed = os.clock() - startTime
        return {
            success = false,
            result = nil,
            error = err,
            elapsed = elapsed,
            timedOut = false,
        }
    end
    
    if coroutine.status(co) ~= "dead" then
        timedOut = true
        local elapsed = os.clock() - startTime
        return {
            success = false,
            result = nil,
            error = "Test coroutine did not complete",
            elapsed = elapsed,
            timedOut = true,
        }
    end
    
    local elapsed = os.clock() - startTime
    
    if elapsed > timeout then
        return {
            success = false,
            result = ret,
            error = "Execution exceeded timeout (" .. formatTime(timeout) .. ")",
            elapsed = elapsed,
            timedOut = true,
        }
    end
    
    return {
        success = true,
        result = ret,
        error = nil,
        elapsed = elapsed,
        timedOut = false,
    }
end

function SafeExec.exists(apiPath)
    local ok, result = pcall(function()
        local parts = string.split(apiPath, ".")
        local current = getfenv and getfenv(0) or _G
        for _, part in ipairs(parts) do
            if current == nil then return nil end
            current = current[part]
        end
        return current
    end)
    if ok then return result end
    return nil
end

function SafeExec.isCallable(value)
    if value == nil then return false end
    if type(value) == "function" then return true end
    local mt = getmetatable(value)
    if mt and mt.__call then return true end
    return false
end

function SafeExec.validateCallable(apiPath)
    local value = SafeExec.exists(apiPath)
    if value == nil then
        return false, "API does not exist: " .. apiPath
    end
    if not SafeExec.isCallable(value) then
        return false, "API exists but is not callable: " .. apiPath
    end
    return true, value
end

local ResultBuilder = {}

function ResultBuilder.new(testName, category)
    return {
        testName = testName,
        category = category,
        status = nil,
        score = 0,
        elapsed = 0,
        error = nil,
        diagnostics = nil,
    }
end

function ResultBuilder.pass(result, score, elapsed, diagnostics)
    result.status = ResultStatus.PASS
    result.score = math.clamp(score or 100, 0, 100)
    result.elapsed = elapsed or 0
    result.diagnostics = diagnostics
    return result
end

function ResultBuilder.fail(result, score, elapsed, reason, diagnostics)
    result.status = ResultStatus.FAIL
    result.score = math.clamp(score or 0, 0, 100)
    result.elapsed = elapsed or 0
    result.error = reason or "Test failed"
    result.diagnostics = diagnostics
    return result
end

function ResultBuilder.error(result, elapsed, err, diagnostics)
    result.status = ResultStatus.ERROR
    result.score = 0
    result.elapsed = elapsed or 0
    result.error = err or "Unknown error"
    result.diagnostics = diagnostics
    return result
end

function ResultBuilder.timeout(result, elapsed, diagnostics)
    result.status = ResultStatus.TIMEOUT
    result.score = 0
    result.elapsed = elapsed or 0
    result.error = "Execution timed out"
    result.diagnostics = diagnostics
    return result
end

function ResultBuilder.unsupported(result, elapsed, reason, diagnostics)
    result.status = ResultStatus.UNSUPPORTED
    result.score = 0
    result.elapsed = elapsed or 0
    result.error = reason or "API not available"
    result.diagnostics = diagnostics
    return result
end

local TestRegistry = {
    categories = {},
    tests = {},
    cleanupHooks = {},
}

function TestRegistry.registerCategory(name, weight, description)
    TestRegistry.categories[name] = {
        name = name,
        weight = weight or 1,
        description = description or "",
        tests = {},
    }
end

function TestRegistry.registerTest(categoryName, testName, fn, weight, description)
    local category = TestRegistry.categories[categoryName]
    if not category then
        error("Category not found: " .. categoryName)
    end
    
    local test = {
        name = testName,
        category = categoryName,
        fn = fn,
        weight = weight or 1,
        description = description or "",
    }
    
    table.insert(category.tests, test)
    table.insert(TestRegistry.tests, test)
end

function TestRegistry.registerCleanup(fn)
    table.insert(TestRegistry.cleanupHooks, fn)
end

function TestRegistry.runCleanup()
    for _, hook in ipairs(TestRegistry.cleanupHooks) do
        pcall(hook)
    end
end

function TestRegistry.getCategories()
    local result = {}
    for _, cat in pairs(TestRegistry.categories) do
        table.insert(result, cat)
    end
    return result
end

-- ═══════════════════════════════════════════════════════════
--  TEST RUNNER
-- ═══════════════════════════════════════════════════════════

local TestRunner = {
    results = {},
    categoryScores = {},
    overallScore = 0,
    startTime = 0,
    endTime = 0,
}

function TestRunner.runAll()
    TestRunner.results = {}
    TestRunner.categoryScores = {}
    TestRunner.startTime = os.clock()
    
    for catName, category in pairs(TestRegistry.categories) do
        local catResults = {}
        local catWeightedSum = 0
        local catWeightTotal = 0
        
        for _, test in ipairs(category.tests) do
            local result = TestRunner.runSingle(test)
            table.insert(catResults, result)
            
            local weight = test.weight
            if result.status == ResultStatus.UNSUPPORTED then
                weight = 0
            end
            catWeightedSum = catWeightedSum + (result.score * weight)
            catWeightTotal = catWeightTotal + weight
            
            table.insert(TestRunner.results, result)
            
            if BenchmarkConfig.CleanupAfterEach then
                TestRegistry.runCleanup()
            end
        end
        
        local catScore = 0
        if catWeightTotal > 0 then
            catScore = catWeightedSum / catWeightTotal
        end
        
        TestRunner.categoryScores[catName] = {
            name = catName,
            score = catScore,
            weight = category.weight,
            results = catResults,
            description = category.description,
        }
    end
    
    local overallWeightedSum = 0
    local overallWeightTotal = 0
    for _, catData in pairs(TestRunner.categoryScores) do
        overallWeightedSum = overallWeightedSum + (catData.score * catData.weight)
        overallWeightTotal = overallWeightTotal + catData.weight
    end
    
    if overallWeightTotal > 0 then
        TestRunner.overallScore = overallWeightedSum / overallWeightTotal
    else
        TestRunner.overallScore = 0
    end
    
    TestRunner.endTime = os.clock()
end

function TestRunner.runSingle(test)
    local result = ResultBuilder.new(test.name, test.category)
    
    local execResult = SafeExec.run(test.fn, BenchmarkConfig.TimeoutPerTest)
    
    if execResult.timedOut then
        return ResultBuilder.timeout(result, execResult.elapsed)
    end
    
    if not execResult.success then
        return ResultBuilder.error(result, execResult.elapsed, execResult.error)
    end
    
    local testResult = execResult.result
    
    if testResult == nil then
        return ResultBuilder.error(result, execResult.elapsed, "Test returned nil result")
    end
    
    if type(testResult) == "table" then
        result.status = testResult.status
        result.score = testResult.score or 0
        result.elapsed = execResult.elapsed
        result.error = testResult.error
        result.diagnostics = testResult.diagnostics
        return result
    end
    
    if testResult == true then
        return ResultBuilder.pass(result, 100, execResult.elapsed)
    elseif testResult == false then
        return ResultBuilder.fail(result, 0, execResult.elapsed, "Test returned false")
    end
    
    return ResultBuilder.error(result, execResult.elapsed, "Unexpected return type: " .. type(testResult))
end

-- ═══════════════════════════════════════════════════════════
--  REPORT RENDERER
-- ═══════════════════════════════════════════════════════════

local ReportRenderer = {}

function ReportRenderer.render()
    local results    = TestRunner.results
    local totalTests = #results
    local passCount  = 0
    local failCount  = 0
    local lines      = {}

    for _, r in ipairs(results) do
        local passed = (r.status == ResultStatus.PASS)
        if passed then
            passCount = passCount + 1
            table.insert(lines, "\xE2\x9C\x85 -- " .. r.testName)
        else
            failCount = failCount + 1
            table.insert(lines, "\xE2\x9D\x8C -- " .. r.testName)
        end
    end

    table.insert(lines, "")
    table.insert(lines, passCount .. "/" .. totalTests)
    table.insert(lines, "")

    local totalTime = TestRunner.endTime - TestRunner.startTime
    table.insert(lines, string.format("finish in %.2f second", totalTime))
    table.insert(lines, "")

    table.insert(lines, "\xE2\x9C\x85 -- Passed")
    table.insert(lines, "\xE2\x9D\x8C -- Failed")

    return table.concat(lines, "\n")
end

    local env = getfenv and getfenv(0) or _G

-- Individual API tests. Each discovered API receives its own result.
-- This is intentionally a discovery + safe invocation check; APIs that
-- require side effects or special context are classified as discovery-only.

for _, group in ipairs(APIRegistry) do
    for _, apiName in ipairs(group.apis) do
        local testName = "API: " .. apiName

        TestRegistry.registerTest("API-Level Capability", testName, function()
            local exists, value = apiExists(apiName)

            if not exists then
                return {
                    status = ResultStatus.UNSUPPORTED,
                    score = 0,
                    error = "API not available",
                    diagnostics = { api = apiName, validation = "discovery" }
                }
            end

            local callable = SafeExec.isCallable(value)

            if callable then
                -- Safe behavioral validation for zero-argument calls only.
                -- Do not call APIs known to have side effects, input injection,
                -- network activity, or resource creation merely for discovery.
                local safeZeroArg = {
                    identifyexecutor = true,
                    getexecutorname = true,
                    getgenv = true,
                    getrenv = true,
                    getgc = true,
                    getmetatable = false,
                    setmetatable = false,
                    getrawmetatable = false,
                    setrawmetatable = false,
                    getnamecallmethod = false,
                    setnamecallmethod = false,
                    hookfunction = false,
                    request = false,
                    http_request = false,
                    ["http.request"] = false,
                    ["WebSocket"] = false,
                    ["websocket.connect"] = false,
                    keypress = false,
                    keyrelease = false,
                    mouse1click = false,
                    mouse1press = false,
                    mouse1release = false,
                    mousemoverel = false,
                    setclipboard = false,
                    toclipboard = false,
                    rconsoleprint = false,
                    rconsolewarn = false,
                    rconsoleerr = false,
                    rconsoleclear = false,
                }

                if safeZeroArg[apiName] == true then
                    local ok, returned = pcall(value)
                    if ok then
                        return {
                            status = ResultStatus.PASS,
                            score = 100,
                            diagnostics = {
                                api = apiName,
                                validation = "discovery+safe-call",
                                returnedType = safeTypeof(returned)
                            }
                        }
                    end

                    return {
                        status = ResultStatus.FAIL,
                        score = 50,
                        error = "API exists but safe zero-argument call failed",
                        diagnostics = { api = apiName, validation = "safe-call" }
                    }
                end

                return {
                    status = ResultStatus.PASS,
                    score = 100,
                    diagnostics = {
                        api = apiName,
                        validation = "discovery+callable"
                    }
                }
            end

            -- Tables/objects such as crypt, Drawing, and WebSocket namespaces
            -- are valid capabilities even when the namespace itself is not callable.
            if type(value) == "table" or type(value) == "userdata" then
                return {
                    status = ResultStatus.PASS,
                    score = 100,
                    diagnostics = {
                        api = apiName,
                        validation = "discovery",
                        valueType = safeTypeof(value)
                    }
                }
            end

            return {
                status = ResultStatus.FAIL,
                score = 0,
                error = "API exists but has an unexpected type",
                diagnostics = {
                    api = apiName,
                    valueType = safeTypeof(value)
                }
            }
        end, 1, "API-level discovery and safe behavioral validation")
    end
end

local function p11Time()
    return os.clock()

local function p11Bounded(fn, iterations)
    local start = p11Time()
    local okCount, errCount = 0, 0
    local firstErr = nil

    for i = 1, iterations do
        local ok, err = pcall(fn, i)
        if ok then
            okCount = okCount + 1
        else
            errCount = errCount + 1
            firstErr = firstErr or tostring(err)
        end
    end

    return okCount, errCount, p11Time() - start, firstErr
end

-- Internal helpers (Phase 13 only, do not conflict with existing helpers)
local function _apiGet(path)
    local parts = string.split and string.split(path,".") or (function()
        local t = {}
        for p in string.gmatch(path,"[^%.]+") do t[#t+1]=p end
        return t
    end)()
    local env = getfenv and getfenv(0) or _G
    local cur = env
    for _, p in ipairs(parts) do
        if cur == nil then return nil end
        local ok, v = pcall(function() return cur[p] end)
        if not ok then return nil end
        cur = v
    end
    return cur
end

local function _isCallable(v)
    if v == nil then return false end
    if type(v) == "function" then return true end
    local ok, mt = pcall(getmetatable, v)
    return ok and mt ~= nil and type(mt.__call) == "function"
end

local function _apiPass(path)
    local v = _apiGet(path)
    if v == nil then
        return {status=ResultStatus.UNSUPPORTED, score=0, error=path.." not found"}
    end
    if not _isCallable(v) then
        return {status=ResultStatus.UNSUPPORTED, score=0, error=path.." not callable"}
    end
    return {status=ResultStatus.PASS, score=100}
end

TestRegistry.registerTest("Individual API - Coroutine","coroutine.yield",function()
    local co=coroutine.create(function()
        coroutine.yield(10); coroutine.yield(20); return 30
    end)
    local ok1,v1=coroutine.resume(co)
    local ok2,v2=coroutine.resume(co)
    local ok3,v3=coroutine.resume(co)
    if ok1 and ok2 and ok3 and v1==10 and v2==20 and v3==30 then return{status=ResultStatus.PASS,score=100}end
    return{status=ResultStatus.FAIL,score=0,error="yield/resume sequence failed: "..tostring(v1)..","..tostring(v2)..","..tostring(v3)}
end,1,"CO-003")

TestRegistry.registerTest("Individual API - Coroutine","coroutine.status",function()
    local co=coroutine.create(function() coroutine.yield() end)
    if coroutine.status(co)~="suspended" then return{status=ResultStatus.FAIL,score=0,error="expected suspended initially"}end
    coroutine.resume(co)
    if coroutine.status(co)~="suspended" then return{status=ResultStatus.FAIL,score=50,error="expected suspended after yield"}end
    coroutine.resume(co)
    if coroutine.status(co)~="dead" then return{status=ResultStatus.FAIL,score=75,error="expected dead after completion"}end
    return{status=ResultStatus.PASS,score=100}
end,1,"CO-004")

TestRegistry.registerTest("Individual API - Coroutine","coroutine.wrap",function()

        if gen()~=1 or gen()~=2 or gen()~=3 then error("wrong values") end
    end); if ok then return{status=ResultStatus.PASS,score=100}end
    return{status=ResultStatus.FAIL,score=0,error="coroutine.wrap failed"}
end,1,"CO-005")
